// DualOS EFI diagnostic filesystem writer.
// Uses only the device handle reported by EFI_LOADED_IMAGE_PROTOCOL.

use alloc::format;
use alloc::string::String;
use core::ffi::c_void;
use core::ptr::null_mut;

use efi::protocol::block_io::BlockIoProtocol;
use efi::protocol::loaded_image::LoadedImageProtocol;
use efi::{efi_call, EfiEntry};
use efi_types::EfiGuid;

#[repr(C)]
struct EfiSimpleFileSystemInterface {
    revision: u64,
    open_volume: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiSimpleFileSystemInterface,
            root: *mut *mut EfiFileProtocol,
        ) -> efi_types::EfiStatus,
    >,
}

#[repr(C)]
struct EfiFileProtocol {
    revision: u64,
    open: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            new_handle: *mut *mut EfiFileProtocol,
            file_name: *const u16,
            open_mode: u64,
            attributes: u64,
        ) -> efi_types::EfiStatus,
    >,
    close: Option<
        unsafe extern "efiapi" fn(this: *mut EfiFileProtocol) -> efi_types::EfiStatus,
    >,
    delete: Option<
        unsafe extern "efiapi" fn(this: *mut EfiFileProtocol) -> efi_types::EfiStatus,
    >,
    read: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            buffer_size: *mut usize,
            buffer: *mut c_void,
        ) -> efi_types::EfiStatus,
    >,
    write: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            buffer_size: *mut usize,
            buffer: *const c_void,
        ) -> efi_types::EfiStatus,
    >,
    get_position: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            position: *mut u64,
        ) -> efi_types::EfiStatus,
    >,
    set_position: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            position: u64,
        ) -> efi_types::EfiStatus,
    >,
    get_info: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,

            buffer_size: *mut usize,
            buffer: *mut c_void,
        ) -> efi_types::EfiStatus,

    set_info: Option<
        unsafe extern "efiapi" fn(
            this: *mut EfiFileProtocol,
            information_type: *const EfiGuid,
            buffer_size: usize,
            buffer: *const c_void,
        ) -> efi_types::EfiStatus,
    >,
    flush: Option<
        unsafe extern "efiapi" fn(this: *mut EfiFileProtocol) -> efi_types::EfiStatus,
    >,
}

const SIMPLE_FILE_SYSTEM_GUID: EfiGuid =
    EfiGuid::from_u64s(0x964e5b22645911d2, 0x8e3900a0c969723b);

const FILE_NAME: [u16; 16] = [
    68, 85, 65, 76, 79, 83, 45, 68, 73, 65, 71, 46, 84, 88, 84, 0,
];

fn build_report(entry: &EfiEntry, source_device: efi::DeviceHandle) -> String {
    let mut report = String::from(
        "DualOS EFI device-path diagnostic
\
         Source: LoadedImageProtocol.DeviceHandle

"
    );

    match crate::utils::get_device_path(entry, source_device) {
        Ok(path) => report.push_str(&format!("LOADED_IMAGE_DEVICE: {path}
")),
        Err(error) => report.push_str(&format!("LOADED_IMAGE_DEVICE_ERROR: {error:?}
")),
    }

    report.push_str(&format!("LOADED_IMAGE_HANDLE: {source_device:?}

"));
    report.push_str("BLOCK_IO_HANDLES:
");

    let boot_services = entry.system_table().boot_services();
    match boot_services.locate_handle_buffer_by_protocol::<BlockIoProtocol>() {
        Ok(handles) => {
            report.push_str(&format!("COUNT: {}
", handles.handles().len()));

            for (index, handle) in handles.handles().iter().enumerate() {
                report.push_str(&format!("
BLOCK #{index} HANDLE: {handle:?}
"));
                match crate::utils::get_device_path(entry, *handle) {
                    Ok(path) => report.push_str(&format!("PATH: {path}
")),
                    Err(error) => report.push_str(&format!("PATH_ERROR: {error:?}
")),
                }
            }
        }
        Err(error) => {
            report.push_str(&format!("BLOCK_IO_ENUMERATION_ERROR: {error:?}
"));
        }
    }

    report.push_str("
END_OF_REPORT
");
    report
}

pub fn write_report(entry: &EfiEntry) -> &'static str {
    let boot_services = entry.system_table().boot_services();

    let loaded_image = match boot_services
        .open_protocol::<LoadedImageProtocol>(entry.image_handle())
    {
        Ok(protocol) => protocol,
        Err(_) => return "FAILED: loaded-image protocol",
    };

    // Restrict filesystem discovery to the device that loaded this EFI image.
    let source_device = loaded_image.device_handle();
    let report = build_report(entry, source_device);

    // SAFETY: The GUID identifies EFI_SIMPLE_FILE_SYSTEM_PROTOCOL. The returned
    // interface is interpreted using the matching UEFI protocol ABI below.
    let filesystem_ptr = match unsafe {
        boot_services.handle_protocol_raw(source_device, &SIMPLE_FILE_SYSTEM_GUID)
    } {
        Ok(pointer) => pointer.as_ptr() as *mut EfiSimpleFileSystemInterface,
        Err(_) => return "FAILED: no filesystem on loaded-image device",
    };

    let mut root: *mut EfiFileProtocol = null_mut();
    // SAFETY: filesystem_ptr is the firmware-provided protocol interface; root is a valid output pointer.
    let open_volume_result = unsafe {
        efi_call!(
            (*filesystem_ptr).open_volume,
            filesystem_ptr,
            &mut root
        )
    };

    if open_volume_result.is_err() || root.is_null() {
        return "FAILED: opening source filesystem";
    }

    let mut file: *mut EfiFileProtocol = null_mut();
    // SAFETY: root is the valid handle returned by OpenVolume; file is a valid output pointer and FILE_NAME is NUL-terminated UTF-16.
    let open_file_result = unsafe {
        efi_call!(
            (*root).open,
            root,
            &mut file,
            FILE_NAME.as_ptr(),
            0x1 | 0x2 | 0x8000_0000_0000_0000,
            0
        )
    };

    if open_file_result.is_err() || file.is_null() {
        // SAFETY: root is the valid file-protocol handle returned by OpenVolume.
        let _ = unsafe { efi_call!((*root).close, root) };
        return "FAILED: creating diagnostic report";
    }

    let report_bytes = report.as_bytes();
    let mut written = report_bytes.len();
    // SAFETY: file is the valid handle returned by Open; REPORT is readable and written is a valid size pointer.
    let write_result = unsafe {
        efi_call!(
            (*file).write,
            file,
            &mut written,
            report_bytes.as_ptr() as *const c_void
        )
    };

    let flush_ok = if write_result.is_ok() && written == report_bytes.len() {
        // SAFETY: file is the valid file-protocol handle returned by Open.
        unsafe { efi_call!((*file).flush, file) }.is_ok()
    } else {
        false
    };

    // SAFETY: file is the valid file-protocol handle returned by Open.
    let _ = unsafe { efi_call!((*file).close, file) };
    // SAFETY: root is the valid file-protocol handle returned by OpenVolume.
    let _ = unsafe { efi_call!((*root).close, root) };

    if write_result.is_err() || written != report_bytes.len() || !flush_ok {
        return "FAILED: writing or flushing report";
    }

    "OK: DUALOS-DIAG.TXT written"
}
