#[cfg(not(test))]
enum TargetOs {
    Android,
    #[cfg(feature = "fuchsia")]
    Fuchsia,
}

#[cfg(not(test))]
fn get_target_os(_entry: &EfiEntry, _disks: &[EfiGblDisk]) -> TargetOs {
    cfg_if! {
        if #[cfg(feature = "fuchsia")] {
            let mut buf = [0u8; 1];
            if _entry
                .system_table()
                .runtime_services()
                .get_variable(&efi::GBL_EFI_VENDOR_GUID, efi::GBL_EFI_OS_BOOT_TARGET_VARNAME, &mut buf)
                .is_ok()
            {
                efi_println!(
                    _entry,
                    "`{}` is set. Proceeding as Fuchsia.",
                    efi::GBL_EFI_OS_BOOT_TARGET_VARNAME
                );
                TargetOs::Fuchsia
            } else if fuchsia_boot::is_fuchsia_gpt(_disks).is_ok() {
                efi_println!(_entry, "Partition layout looks like Fuchsia. Proceeding as Fuchsia");
                TargetOs::Fuchsia
            } else {
                efi_println!(_entry, "Proceeding as Android");
                TargetOs::Android
            }
        } else {
            TargetOs::Android
        }
    }
}

/// GBL EFI application logic entry point.
#[cfg(not(test))]
#[allow(unreachable_code, unused_variables)]
pub fn app_main(entry: EfiEntry) -> Result<Infallible> {
    efi_println!(entry, "**** Generic Bootloader (GBL) ****");
    efi_println!(entry, "Fingerprint: {}", format_build_fingerprint!());
    efi_println!(entry, "VCS info: {}", format_build_vcs_info!());
    if let Ok(v) = loaded_image_path(&entry) {
        efi_println!(entry, "Image path: {v}");
    }

    // Read-only console mode diagnostic.
    match entry.system_table().con_out() {
        Ok(console) => {
            let mode_info = console.interface().mode;
            if mode_info.is_null() {
                efi_println!(entry, "Console mode information unavailable.");
            } else {
                // SAFETY: UEFI provides this pointer as part of the live
                // Simple Text Output Protocol.
                let (max_mode, current_mode) = unsafe {
                    ((*mode_info).max_mode, (*mode_info).mode)
                };
                efi_println!(
                    entry,
                    "Console current mode: {current_mode}; available modes: {max_mode}"
                );

                for mode in 0..max_mode.max(0) as usize {
                    match console.query_mode(mode) {
                        Ok((cols, rows)) => {
                            efi_println!(
                                entry,
                                "Console mode {mode}: {cols} columns x {rows} rows"
                            );
                        }
                        Err(e) => {
                            efi_println!(
                                entry,
                                "Console mode {mode}: query failed: {e:?}"
                            );
                        }
                    }
                }
            }
        }
        Err(e) => efi_println!(entry, "Console protocol unavailable: {e:?}"),
    }

    let report_status = diagnostic_fs::write_report(&entry);
    efi_println!(entry, "Filesystem report: {report_status}");
    efi_println!(entry, "Diagnostic complete. Halting before OS boot.");
    loop {
        core::hint::spin_loop();
    }
}
