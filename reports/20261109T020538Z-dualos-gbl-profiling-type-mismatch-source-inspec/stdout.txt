/// Finds and returns all EFI devices supporting either EFI_BLOCK_IO or EFI_BLOCK_IO2 protocol.
#[profile(backend = EfiProfileBackend::new(efi_entry))]
pub fn find_block_devices(efi_entry: &EfiEntry) -> Result<Vec<EfiGblDisk<'_>>, Error> {
    let bs = efi_entry.system_table().boot_services();
    let block_dev_handles = bs.locate_handle_buffer_by_protocol::<BlockIoProtocol>()?;

    // GBL is SD-only: identify the physical disk that contains this EFI image.
    let loaded_image = bs.open_protocol::<LoadedImageProtocol>(efi_entry.image_handle())?;
    let source_partition = loaded_image.device_handle();
    let source_path = bs.open_protocol::<DevicePathProtocol>(source_partition)?;

    let source_block_io = bs.open_protocol::<BlockIoProtocol>(source_partition)?;
    // SAFETY: this code always executes at EFI_TPL_APPLICATION.
    let source_media = unsafe {
        source_block_io.with_lock::<EFI_TPL_APPLICATION, _>(efi_entry, || {
            source_block_io.media().unwrap().clone()
        })
    };
    if !source_media.logical_partition {
        let status = "FAIL: loaded-image device handle is not a logical partition";
        efi_println!(efi_entry, "DUALOS SD-ONLY FILTER: {status}");
        #[cfg(not(test))]
        write_dualos_discovery_report(efi_entry, status);
        return Err(Error::InvalidInput);
    }

    let mut source_disk_handle = None;
    for handle in block_dev_handles.handles() {
        let candidate_path = match bs.open_protocol::<DevicePathProtocol>(*handle) {
            Ok(path) => path,
            Err(_) => continue,
        };
        if !candidate_path.is_parent_of(&source_path) {
            continue;
        }

        let candidate_block_io = match bs.open_protocol::<BlockIoProtocol>(*handle) {
            Ok(protocol) => protocol,
            Err(_) => continue,
        };
        // SAFETY: this code always executes at EFI_TPL_APPLICATION.
        let candidate_media = unsafe {
            candidate_block_io.with_lock::<EFI_TPL_APPLICATION, _>(efi_entry, || {
                candidate_block_io.media().unwrap().clone()
            })
        };
        if candidate_media.logical_partition {
            continue;
        }

        if source_disk_handle.replace(*handle).is_some() {
            let status = "FAIL: multiple physical parent disks matched the loaded-image partition";
            efi_println!(efi_entry, "DUALOS SD-ONLY FILTER: {status}");
            #[cfg(not(test))]
            write_dualos_discovery_report(efi_entry, status);
            return Err(Error::InvalidInput);
        }
    }

    let source_disk_handle = match source_disk_handle {
        Some(handle) => handle,
        None => {
            let status = "FAIL: no physical parent disk found for the loaded-image partition";
            efi_println!(efi_entry, "DUALOS SD-ONLY FILTER: {status}");
            #[cfg(not(test))]
            write_dualos_discovery_report(efi_entry, status);
            return Err(Error::InvalidInput);
        }
    };

    let selected_disk_path = crate::utils::get_device_path(efi_entry, source_disk_handle)
        .unwrap_or_else(|error| alloc::format!("UNAVAILABLE: {error:?}"));

    efi_println!(
        efi_entry,
        "DUALOS FILTER: PASS — all other physical disks excluded from GBL discovery."
    );
